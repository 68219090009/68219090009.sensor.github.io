การทดลองนี้ใช้ตรวจสอบว่าโมดูล KY-032 สามารถตรวจจับสิ่งกีดขวางด้วยคลื่นอินฟราเรด และส่งสถานะดิจิทัลไปแสดงผลบน Serial Monitor ได้ถูกต้องก่อนนำไปใช้กับหุ่นยนต์หรือระบบควบคุมอัตโนมัติ

1. วัตถุประสงค์

ยืนยันว่าการตรวจจับสิ่งกีดขวางระยะสั้นด้วยคลื่นอินฟราเรดสามารถอ่านค่าแบบดิจิทัลได้ถูกต้อง และแสดงผลการตรวจจับผ่าน Serial Monitor อย่างชัดเจน เพื่อประเมินระยะและประสิทธิภาพของเซ็นเซอร์

อุปกรณ์ที่ใช้

  • Arduino Uno หรือ ESP32 และโมดูล KY-032
  • สาย Jumper แบบ Female to Male
  • คอมพิวเตอร์ที่ติดตั้ง Arduino IDE หรือ PlatformIO
  • Serial Monitor ตั้งความเร็วที่ 9600 Baud Rate
  • กระดาษสีขาวหรือวัตถุผิวเรียบสำหรับทดสอบ

2. หลักการทดสอบ

  1. ต่อสายไฟเลี้ยงและเชื่อมขา OUT เข้ากับไมโครคอนโทรลเลอร์
  2. กำหนดขา GPIO เป็นโหมดรับข้อมูลดิจิทัลด้วย INPUT
  3. อ่านสถานะด้วยคำสั่ง digitalRead() ในวงรอบการทำงาน
  4. ส่งผลสถานะไปแสดงบน Serial Monitor ที่ความเร็ว 9600
  5. นำวัตถุมาบังด้านหน้า และปรับระยะด้วยตัวปรับ R5 บนโมดูล

การต่อวงจร Arduino Uno กับ KY-032

3. ตัวอย่างโปรแกรมทดสอบ

โปรแกรมจะอ่านค่าจากขา D2 แล้วแสดงค่า Raw Signal พร้อมสถานะการตรวจจับทุก 300 มิลลิวินาที

const int sensorPin = 2;

void setup() {
  Serial.begin(9600);
  pinMode(sensorPin, INPUT);
  Serial.println("เริ่มต้นการทดสอบ KY-032");
}

void loop() {
  int sensorState = digitalRead(sensorPin);

  Serial.print("Raw Signal: ");
  Serial.print(sensorState);
  if (sensorState == LOW) {
    Serial.println(" | พบสิ่งกีดขวาง");
  } else {
    Serial.println(" | เส้นทางว่าง");
  }
  delay(300);
}

4. การวิเคราะห์ผลลัพธ์

  • ไม่พบวัตถุ: ไฟ LED ดับ และแสดง Raw Signal: 1 | เส้นทางว่าง
  • พบสิ่งกีดขวาง: ไฟ LED ติด และแสดงค่า Raw Signal เป็น 0
  • แก้ไขเมื่อพบสัญญาณค้าง: หมุน R5 ทวนเข็มนาฬิกาเพื่อลดความไว และตรวจสอบ Jumper ที่ขา EN

สรุป: การทดสอบผ่าน Serial Monitor ช่วยยืนยันการทำงานของวงจรอินฟราเรดและการรับสัญญาณดิจิทัล สามารถนำผลไปใช้ควบคุมมอเตอร์ สร้างหุ่นยนต์หลบสิ่งกีดขวาง หรือประยุกต์กับระบบนับชิ้นงานได้
